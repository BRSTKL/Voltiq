import Head from "next/head";
import Link from "next/link";
import { useMemo } from "react";
import { useRouter } from "next/router";
import { useSession } from "next-auth/react";
import { ArrowRight, CheckCircle2, Sparkles } from "lucide-react";

function formatPlan(plan) {
  if (plan === "ENTERPRISE") {
    return "Enterprise";
  }

  if (plan === "PRO") {
    return "Pro";
  }

  return "Free";
}

  const router = useRouter();
  const { data: session, status } = useSession();

  const currentPlan = session?.user?.plan ?? "FREE";
  const showUpgradeBanner = useMemo(
    () => router.query.upgraded === "true",
    [router.query.upgraded]
  );

            <section className="rounded-3xl border border-emerald-500/30 bg-emerald-500/10 p-6">
              <div className="flex items-start gap-4">
                <div className="flex h-12 w-12 items-center justify-center rounded-2xl bg-emerald-500/20">
                  <CheckCircle2 className="h-6 w-6 text-emerald-300" />
                </div>
                <div className="space-y-2">
                  <p className="text-sm font-semibold uppercase tracking-[0.2em] text-emerald-200">
                    Upgrade confirmed
                  </p>

                    Stripe checkout tamamlandi. Planin kisa sure icinde guncellenecek.
                  </h1>
                  <p className="text-sm leading-6 text-emerald-50/85">
                    Webhook geldigi anda hesabin Pro veya Enterprise olarak isaretlenecek.
                    Bu sayfayi yenileyerek guncel durumu gorebilirsin.
                  </p>
                </div>
              </div>
            </section>
          ) : null}

          <section className="rounded-[2rem] border border-white/10 bg-[radial-gradient(circle_at_top,_rgba(16,185,129,0.14),_transparent_42%),linear-gradient(180deg,rgba(17,24,39,0.95),rgba(3,7,18,1))] p-8 sm:p-10">
            <div className="flex flex-col gap-8 lg:flex-row lg:items-end lg:justify-between">
              <div className="space-y-4">
                <span className="inline-flex items-center rounded-full border border-white/10 bg-white/5 px-3 py-1 text-xs font-semibold uppercase tracking-[0.2em] text-emerald-200">
